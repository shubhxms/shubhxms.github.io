Returns nil when the entry has no cover. */}} {{- $out := dict }} {{- with .cover }} {{- with resources.Get (strings.TrimPrefix "/" .) }} {{- $thumb := .Fill "240x360 Center webp q80" }} {{- $colour := index .Colors 0 }} {{- $out = dict "thumb" $thumb.RelPermalink "colour" $colour.ColorHex "light" (lt $colour.Luminance 0.45) }} {{- else }} {{- errorf "shelf: cover %q is not under static/images/shelf" . }} {{- end }} {{- end }} {{- return $out }}
